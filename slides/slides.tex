\section{Motivation}

\begin{frame}{Why Lie Groups? Why Quantum Spin?}
\begin{block}{Puzzle}
Introductory quantum mechanics presents spin heuristically, as an
``intrinsic angular momentum'' with mysterious half-integer values.
Where does this come from?
\end{block}

\pause
\vspace{0.5em}
\textbf{Approach.} Build the answer from the ground up:
\begin{enumerate}
\item Topology $\to$ smooth manifolds: the geometric language.
\item Lie groups = smooth manifolds + group structure: continuous symmetries.
\item Representation theory: how abstract groups act on vector spaces.
\item $\SU(2)$ and $\SO(3)$: the rotation group and its double cover.
\item Classification of irreps $\to$ integer \& half-integer spin.
\end{enumerate}

\pause
\vspace{0.3em}
Topology and abstract algebra $\longrightarrow$ quantum physics.
\end{frame}

\section{Topological Spaces}

\begin{frame}{Topological Spaces}
\begin{Def}{Definition (topological space)}
A \textbf{topological space} is a pair $(X,\T)$, where $X$ is a set and
$\T\subseteq\P(X)$ satisfies:
\begin{enumerate}
\item (Trivial bounds) $\varnothing\in\T$ and $X\in\T$.
\item (Arbitrary unions)
$\forall\set{U_\alpha}_{\alpha\in\I}\subseteq\T
:\bigcup_{\alpha\in\I}U_\alpha\in\T$.
\item (Finite intersections)
$\forall U,V\in\T:U\cap V\in\T$.
\end{enumerate}
\end{Def}

\pause
\vspace{0.5em}
\textbf{Significance.} A topology tells us what ``closeness'' means
without needing a distance function. Open sets are the primitive notion
of ``nearby'': they generalise open intervals in $\R$.

\pause
\vspace{0.3em}
\textbf{Constructions.} Product topology, subspace topology, metric
topologies (e.g.\ the Euclidean topology on $\R^n$).
\end{frame}

\section{Smooth Manifolds}

\begin{frame}{From Topological to Smooth Manifolds}
\begin{Def}{Definition (topological manifold)}
A \textbf{topological manifold} is a space $(M,\T)$ that is Hausdorff,
second-countable, and locally Euclidean of dimension $n$.
\end{Def}

\pause
\begin{Def}{Definition (smooth manifold)}
A \textbf{smooth manifold} is a triple $(M,\T,\A)$ where:
\begin{enumerate}
\item $(M,\T)$ is a topological manifold.
\item $\A=\set{(U_\alpha,\phi_\alpha)}_{\alpha\in\I}$ is a
\textbf{smooth atlas}: charts cover $M$, and transition maps
$\phi_\beta\circ\phi_\alpha^{-1}$ are $C^\infty$.
\item $\A$ is \textbf{maximal} with respect to smooth compatibility.
\end{enumerate}
\end{Def}

\pause
\vspace{0.3em}
\textbf{Idea.} Charts are local coordinate systems. Smoothness means
change-of-coordinates is infinitely differentiable. A \textbf{diffeomorphism}
is a smooth bijection with smooth inverse: manifolds that are ``the same.''
\end{frame}

\section{Germs and Stalks}

\begin{frame}{Local Smooth Representatives}
\begin{Def}{Definition (local smooth representative)}
Let $(M,\T,\A)$ be a smooth manifold, $p\in M$.
The \textbf{set of local smooth representatives at $p$} is
\[
\Rep_p(M,\T,\A)\df\set{(U,f)\st p\in U\in\T\tand
f:U\to\R~\text{is smooth}}.
\]
A pair $(U,f)\in\Rep_p(M,\T,\A)$ is a smooth map defined on some
open neighbourhood $U$ of $p$.
\end{Def}

\pause
\vspace{0.5em}
\textbf{Why local representatives?} On a manifold, smooth maps may only
be defined locally (e.g.\ on a chart domain). To work algebraically at a
point, we need all functions defined on \emph{any} neighbourhood
of that point, not just globally-defined ones.
\end{frame}

\begin{frame}{Germs: Equivalence}
\begin{block}{Motivation}
To define tangent vectors algebraically, we need to know when two locally
defined smooth maps are ``the same near a point'', even if they differ
far away. This is captured by \textbf{germs}.
\end{block}

\pause
\begin{Def}{Definition (germ equivalence)}
Let $p\in M$ and $(U,f),(V,g)\in\Rep_p(M,\T,\A)$.
$(U,f)$ and $(V,g)$ are \textbf{germ equivalent at $p$}, denoted
$(U,f)\sim_p(V,g)$, iff
\[
\exists W\in\T:\quad p\in W\subseteq U\cap V\tand f|_W=g|_W.
\]
\end{Def}

\pause
\vspace{0.5em}
\textbf{Intuition.} Two functions are germ equivalent if they agree on
some neighbourhood of $p$. What happens far from $p$ does not matter.
\end{frame}

\begin{frame}{Germs: Germ and Stalk}
\begin{Def}{Definition (germ)}
Let $p\in M$ and $(U,f)\in\Rep_p(M,\T,\A)$.
The \textbf{$(U,f)$-germ at $p$} is the equivalence class
\[
\SqBr{(U,f)}_p\df\set{(V,g)\in\Rep_p(M,\T,\A)\st (U,f)\sim_p(V,g)}.
\]
\end{Def}

\pause
\begin{Def}{Definition (stalk)}
The \textbf{stalk of smooth maps at $p$} is the quotient
\[
\begin{gathered}
C_p^\infty(M,\T,\A)\df\Rep_p(M,\T,\A)/\sim_p\\
=\set{\SqBr{(U,f)}_p\st (U,f)\in\Rep_p(M,\T,\A)}.
\end{gathered}
\]
\end{Def}

\pause
\vspace{0.3em}
\textbf{Structure.} The stalk $C_p^\infty(M,\T,\A)$ is a commutative
associative unital $\R$-algebra under pointwise operations on germs.
\end{frame}

\section{Tangent Spaces}

\begin{frame}{Tangent Spaces via Derivations}
\begin{Def}{Definition (derivation at a point)}
Let $p\in M$. A map $v:C_p^\infty(M,\T,\A)\to\R$ is a
\textbf{derivation at $p$} iff:
\begin{enumerate}
\item ($\R$-Linearity) $v(\alpha+\lambda\beta)=v(\alpha)+\lambda\,v(\beta)$.
\item (Leibniz Rule) $v(\alpha\times\beta)=v(\alpha)\,\beta(p)+\alpha(p)\,v(\beta)$.
\end{enumerate}
\end{Def}

\pause
\begin{Def}{Definition (tangent space)}
The \textbf{tangent space at $p$} is
\[
T_pM\df\set{v:C_p^\infty(M,\T,\A)\to\R\st
v~\text{is a derivation at}~p}.
\]
$T_pM$ is an $\R$-vector space with $\dim(T_pM)=\dim(M)$.
\end{Def}

\pause
\vspace{0.5em}
\textbf{Motivation.} For $\R^n$, a tangent vector $\bm v$ acts by the
directional derivative: $\bm v(f)=\sum_i v^i\frac{\p f}{\p x^i}\big|_p$.
The derivation formalism generalises this to any manifold without an
embedding.
\end{frame}

\begin{frame}{The Differential}
\begin{Def}{Definition (differential)}
Let $F:M\to N$ be smooth, $p\in M$. The \textbf{differential of $F$ at $p$}
is the linear map $\dd F_p:T_pM\to T_{F(p)}N$ defined by
\[
\forall v\in T_pM,\;\forall \SqBr{f}_{F(p)}\in C_{F(p)}^\infty N:\quad
\FUNC{(\dd F_p(v))}{\SqBr{f}_{F(p)}}\df v\Br{\SqBr{f\circ F}_p}.
\]
\end{Def}

\pause
\begin{block}{Chain rule}
For smooth maps $F:M\to N$ and $G:N\to P$,
\[
\dd(G\circ F)_p = \dd G_{F(p)}\circ \dd F_p.
\]
\end{block}

\pause
\vspace{0.5em}
\textbf{Interpretation.} The differential is the ``best linear approximation''
to a smooth map, generalising the Jacobian matrix from multivariable calculus.
\end{frame}

\section{Smooth Vector Fields}

\begin{frame}{Smooth Vector Fields: Definition}
\begin{block}{Motivation}
A tangent vector lives at a single point. A \textbf{vector field} assigns a
tangent vector to \emph{every} point, smoothly. Algebraically, it is a
derivation on the algebra of global smooth maps.
\end{block}

\pause
\begin{Def}{Definition (smooth vector field)}
Let $(M,\T,\A)$ be a smooth manifold.
A map $X:C^\infty(M,\T,\A)\to C^\infty(M,\T,\A)$ is a
\textbf{smooth vector field} iff:
\begin{enumerate}
\item ($\R$-Linearity) $X$ is $\R$-linear.
\item (Leibniz Rule) $\forall f,g\in C^\infty(M,\T,\A):\;
X(f\times g)=X(f)\times g+f\times X(g)$.
\end{enumerate}
The \textbf{set of smooth vector fields} is
\[
\X(M,\T,\A)\df\set{X\in\L\Br{C^\infty(M,\T,\A)}\st
X~\text{is a smooth vector field}}.
\]
\end{Def}

\pause
\vspace{0.3em}
\textbf{Insight.} A smooth vector field is a \textbf{global derivation} on
$C^\infty(M)$. It takes a smooth map and produces another smooth map
; it is the derivative along $X$, satisfying the Leibniz rule. $\X(M)$ is an
$\R$-vector space under pointwise operations.
\end{frame}

\begin{frame}{Smooth Vector Fields: Evaluation at a Point}
\begin{block}{Question}
A smooth vector field $X$ acts on \emph{global} functions.
How do we recover a tangent vector $X_p\in T_pM$ at a single point?
\end{block}

\pause
\begin{Def}{Definition (value of a vector field at a point)}
Let $X\in\X(M,\T,\A)$ and $p\in M$. The \textbf{value of $X$ at $p$},
denoted $\eval{X}_p:C_p^\infty(M,\T,\A)\to\R$, is
\[
\forall\SqBr{(U,f)}_p\in C_p^\infty(M,\T,\A):\quad
\eval{X}_p\Br{\SqBr{(U,f)}_p}\df\FUNC{X\Br{\tilde f}}{p},
\]
where $\tilde f\in C^\infty(M,\T,\A)$ is any global representative of
the germ ; existence follows from the Germ Extension Lemma.
\end{Def}

\pause
\begin{Def}{Definition (evaluation map)}
The \textbf{evaluation map at $p$} is
\[
\operatorname{ev}_p:\X(M,\T,\A)\to T_p(M,\T,\A),\qquad
\operatorname{ev}_p(X)\df\eval{X}_p.
\]
\end{Def}
\end{frame}

\begin{frame}{Smooth Vector Fields: Properties of Evaluation}
\begin{lemma}[Value of a vector field is a derivation]
$\eval{X}_p\in T_p(M,\T,\A)$: the value of a smooth vector field
at a point is a tangent vector.
\end{lemma}

\pause
\begin{lemma}[Evaluation map is linear]
$\operatorname{ev}_p:\X(M,\T,\A)\to T_p(M,\T,\A)$
is a linear transformation:
\[
\operatorname{ev}_p(X\pwplus\lambda\pwcdot Y)
=\operatorname{ev}_p(X)\pwplus\lambda\pwcdot\operatorname{ev}_p(Y).
\]
\end{lemma}

\pause
\vspace{0.5em}
\textbf{Fact.} Every tangent vector is the value of \emph{some} smooth
vector field at $p$. Locality lemma: if $h$ vanishes on a neighbourhood
of $p$, then $\FUNC{X(h)}{p}=0$.
\end{frame}

\begin{frame}{Vector Fields Form a Lie Algebra}
\begin{Thm}{Theorem (Lie algebra of vector fields)}
Let $(M,\T,\A)$ be a smooth manifold.
Define the \textbf{commutator} of $X,Y\in\X(M,\T,\A)$ by
$[X,Y]_\circ\df X\circ Y-Y\circ X$, i.e.,
\[
\forall f\in C^\infty(M,\T,\A):\quad
\Func{[X,Y]_\circ}{f}=X(Y(f))-Y(X(f)).
\]
Then $[X,Y]_\circ\in\X(M,\T,\A)$ and
$\Br{\X(M,\T,\A),\set{+,-,\bm0,\cdot,[\bullet,\bullet]_\circ}}$
is a Lie $\R$-algebra.
\end{Thm}

\pause
\vspace{0.5em}
\textbf{Significance.} The commutator of two vector fields measures the
failure of their flows to commute. It encodes the geometry of the manifold
in an algebraic structure.
\end{frame}

\section{Algebraic Structures}

\begin{frame}{Groups, Algebras, and Lie Algebras}
\begin{Def}{Definition (group)}
A set $G$ with an associative binary operation $\mu$, an identity $e$,
and inverses $\iota$. A group is the algebraic abstraction of \textbf{symmetry}.
\end{Def}

\pause
\begin{Def}{Definition ($\F$-algebra)}
A vector space $\A$ over $\F$ together with a bilinear multiplication
$\A\times\A\to\A$. (Examples: $\R^{n,n}$ with matrix multiplication,
$C^\infty(M)$ with pointwise multiplication.)
\end{Def}

\pause
\begin{Def}{Definition (Lie algebra)}
An $\F$-vector space $\g$ with a bilinear bracket
$\SqBr{\bullet,\bullet}:\g\times\g\to\g$ satisfying:
\begin{enumerate}
\item (Skew-symmetry) $\SqBr{Y,X}=-\SqBr{X,Y}$
\item (Jacobi identity)
$\SqBr{X,\SqBr{Y,Z}}+\SqBr{Y,\SqBr{Z,X}}+\SqBr{Z,\SqBr{X,Y}}=\bm0$
\end{enumerate}
\end{Def}

\pause
\vspace{0.3em}
\textbf{Example.} $\R^3$ with the cross product is a Lie algebra.
The commutator $\SqBr{\bm A,\bm B}=\bm A\bm B-\bm B\bm A$ on matrices
makes any matrix space a Lie algebra.
\end{frame}

\section{Lie Groups}

\begin{frame}{Lie Groups}
\begin{Def}{Definition (Lie group)}
A \textbf{Lie group} is a smooth manifold $(G,\T,\A)$ that is also a group
$\Br{G,\set{\mu,\iota,e}}$ such that:
\begin{enumerate}
\item $\mu:G\times G\to G$ (multiplication) is smooth.
\item $\iota:G\to G$ (inversion) is smooth.
\end{enumerate}
\end{Def}

\pause
\vspace{0.5em}
\textbf{Synthesis.} A Lie group unifies geometry (smooth manifold)
with algebra (group). This allows us to study continuous symmetries
using both differential calculus and group theory.

\pause
\vspace{0.3em}
\textbf{Examples.}
\begin{itemize}
\item $(\R^n,+)$: the translation group.
\item $\GL(n,\R)$: invertible $n\times n$ real matrices  of dimension $n^2$.
\item $\SO(n)$: rotations in $\R^n$.
\item $\SU(n)$: unitary transformations with determinant $1$.
\end{itemize}
\end{frame}

\begin{frame}{Left-Invariant Vector Fields}
\begin{Def}{Definition (left translation)}
For $g\in G$, $L_g:G\to G$ is defined by $L_g(h)\df gh$.
$L_g$ is a diffeomorphism.
\end{Def}

\pause
\begin{Def}{Definition (left-invariant vector field)}
A smooth vector field $X\in\X(G,\T,\A)$ is \textbf{left-invariant} if
\[
\forall g\in G,\;\forall f\in C^\infty(G):\quad
X(f\circ L_g)=X(f)\circ L_g.
\]
\end{Def}

\pause
\vspace{0.5em}
\textbf{Meaning.} A left-invariant vector field is completely
determined by its value at the identity $e$. Translating the vector
at $e$ by $L_g$ gives the vector at any point $g$.
\end{frame}

\begin{frame}{The Lie Algebra of a Lie Group}
\begin{block}{Evaluation at the identity}
Recall the evaluation map: for any smooth vector field $X$,
$\eval{X}_e$ is its value at the identity $e\in G$. The map
\[
\operatorname{ev}_e:\X_L(G)\to T_eG,\qquad
\operatorname{ev}_e(X)\df\eval{X}_e,
\]
takes a left-invariant vector field and returns its tangent vector at $e$.
This map is a \textbf{linear isomorphism}.
\end{block}

\pause
\begin{Thm}{Theorem (tangent space at identity is a Lie algebra)}
Let $G$ be a Lie group. Define $\g = T_eG$.
For each $v\in\g$, there exists a \textbf{unique} left-invariant smooth
vector field $X^v$ with $\eval{X^v}_e=v$. Define the bracket on $\g$ by
\[
\SqBr{v,w}_\g\df\eval{\SqBr{X^v,X^w}_\circ}_e.
\]
Then $\Br{\g,\SqBr{\bullet,\bullet}_\g}$ is a Lie $\R$-algebra with
$\dim(\g)=\dim(G)$.
\end{Thm}

\pause
\textbf{Bridge.} Every Lie group gives rise to a Lie algebra
, its tangent space at the identity,. The Lie algebra captures the
\textbf{infinitesimal} structure of the group. We write $\Lie(G)$.
\end{frame}

\section{Matrix Lie Groups}

\begin{frame}{The Rotation Group \texorpdfstring{$\SO(3)$}{SO(3)}}
\begin{Ex}{Example (the special orthogonal group)}
\[
\begin{aligned}
\SO(3)&=\set{\bm R\in\R^{3,3}\st\bm R\transp\bm R=\bm I_3,\;
\det(\bm R)=1};\\[0.3em]
\so(3)&=T_{\bm I_3}\SO(3)=\set{\bm R\in\R^{3,3}\st\bm R\transp=-\bm R}.
\end{aligned}
\]
$\dim(\so(3))=\dim(\SO(3))=3$, with basis satisfying
$\SqBr{\bm L_i,\bm L_j}=\eps_{ijk}\bm L_k$:
\[
\bm L_1=\mqty{0&0&0\\0&0&-1\\0&1&0},\;
\bm L_2=\mqty{0&0&1\\0&0&0\\-1&0&0},\;
\bm L_3=\mqty{0&-1&0\\1&0&0\\0&0&0}.
\]
\end{Ex}

\pause
\vspace{0.3em}
\textbf{Remark.} $\SqBr{\bm L_i,\bm L_j}=\eps_{ijk}\bm L_k$:
the same structure as the cross product on $\R^3$.
\end{frame}

\begin{frame}{The Special Unitary Group \texorpdfstring{$\SU(2)$}{SU(2)}}
\begin{Ex}{Example (the special unitary group)}
\[
\begin{aligned}
\SU(2)&=\set{\bm U\in\C^{2,2}\st\bm U\herm\bm U=\bm I_2,\;
\det(\bm U)=1};\\[0.3em]
\su(2)&=T_{\bm I_2}\SU(2)
=\set{\bm U\in\C^{2,2}\st\bm U\herm=-\bm U,\;\tr(\bm U)=0}.
\end{aligned}
\]
$\dim(\su(2))=\dim(\SU(2))=3$, with basis satisfying
$\SqBr{\bm T_i,\bm T_j}=\eps_{ijk}\bm T_k$:
\[
\bm T_1=\frac12\mqty{0&-\ii\\-\ii&0},\;
\bm T_2=\frac12\mqty{0&-1\\1&0},\;
\bm T_3=\frac12\mqty{-\ii&0\\0&\ii}.
\]
\end{Ex}

\pause
\vspace{0.3em}
\textbf{Remark.} $\SqBr{\bm T_i,\bm T_j}=\eps_{ijk}\bm T_k$:
\textbf{identical} to $\so(3)$. Hence $\so(3)\cong\su(2)$ as Lie algebras.
\end{frame}

\section{The Double Cover \texorpdfstring{$\SU(2)\to\SO(3)$}{SU(2)→SO(3)}}

\begin{frame}{The Double Cover \texorpdfstring{$\Phi:\SU(2)\to\SO(3)$}{Φ: SU(2)→SO(3)}}
\begin{block}{Construction}
For $\bm U\in\SU(2)$, define $\Phi(\bm U):\su(2)\to\su(2)$ by
\[
\FUNC{\Phi(\bm U)}{\bm X}\df\bm U\bm X\bm U\herm.
\]
Using the basis $(\bm T_1,\bm T_2,\bm T_3)$, identify
$\su(2)\cong\R^3$. Then the matrix $\SqBr{\Phi(\bm U)}_\beta^\beta$
lies in $\SO(3)$.
\end{block}

\pause
\begin{Thm}{Theorem (double cover)}
$\Phi:\SU(2)\to\SO(3)$ is:
\begin{enumerate}
\item A smooth group homomorphism.
\item Surjective.
\item Has kernel $\ker(\Phi)=\set{\pm\bm I_2}$ .
\end{enumerate}
Thus $\SU(2)$ is a \textbf{double cover} of $\SO(3)$.
\end{Thm}

\pause
\vspace{0.3em}
\textbf{Two-to-one.} Both $\bm U$ and $-\bm U$ give the same rotation.
A rotation by angle $\theta$ about axis $\hat{\bm n}$ lifts to
$\pm\cos(\theta/2)\bm I_2\mp\ii\sin(\theta/2)(\hat{\bm n}\cdot\bm\sigma)$.
\end{frame}

\begin{frame}{The Double Cover: Geometric Picture}
\begin{center}
\begin{tikzpicture}[scale=1.5]
  % SU(2) as S^3
  \draw[fill=blue!15] (0,0) circle (1.2);
  \node at (0,1.5) {$\SU(2)\cong S^3$};
  \node at (0,0) {\small (simply connected)};
  \draw[fill=black] (0,0.3) circle (0.04) node[right] {$+\bm I_2$};
  \draw[fill=black] (0,-0.3) circle (0.04) node[right] {$-\bm I_2$};

  % Arrow
  \draw[->,thick] (1.5,0) -- (3.5,0) node[midway,above] {$\Phi$};

  % SO(3) as RP^3
  \draw[fill=red!15] (5,0) circle (1.2);
  \node at (5,1.5) {$\SO(3)\cong \R P^3$};
  \node at (5,0.3) {\small (not simply connected)};
  \draw[fill=black] (5,0) circle (0.04) node[right] {$\bm I_3$};
\end{tikzpicture}
\end{center}

\pause
\vspace{0.5em}
\textbf{Interpretation.} $\SU(2)\cong S^3$ , the $3$-sphere, is
simply connected: it is the \textbf{universal cover} of $\SO(3)\cong\R P^3$.
The two-to-one map identifies antipodal points on $S^3$.
\end{frame}

\section{Complexification and Cartan-Weyl Basis}

\begin{frame}{Complexification \texorpdfstring{$\su(2)_\C\cong\mathfrak{sl}(2,\C)$}{su(2)_C ≅ sl(2;C)}}
\begin{Def}{Definition (special linear Lie algebra)}
$\mathfrak{sl}(2,\C)\df\set{\bm A\in\C^{2,2}\st\tr(\bm A)=0}$,
with bracket $\SqBr{\bm A,\bm B}=\bm A\bm B-\bm B\bm A$.
\end{Def}

\pause
\begin{Def}{Definition (Cartan-Weyl basis)}
The \textbf{Cartan-Weyl basis} $\set{\bm H,\bm E,\bm F}$ of $\mathfrak{sl}(2,\C)$:
\[
\bm H=\mqty{1&0\\0&-1},\quad
\bm E=\mqty{0&1\\0&0},\quad
\bm F=\mqty{0&0\\1&0}.
\]
Satisfying: $\SqBr{\bm H,\bm E}=2\bm E,\;
\SqBr{\bm H,\bm F}=-2\bm F,\;
\SqBr{\bm E,\bm F}=\bm H$.
\end{Def}

\pause
\begin{Thm}{Theorem (\texorpdfstring{$\su(2)_\C\cong\mathfrak{sl}(2,\C)$}{su(2)_C ≅ sl(2;C)})}
The complexification $\su(2)_\C=\su(2)\otimes_\R\C$ is isomorphic to
$\mathfrak{sl}(2,\C)$ via mutually-inverse maps
$\Phi:\mathfrak{sl}(2,\C)\to\su(2)_\C$ and
$\Psi:\su(2)_\C\to\mathfrak{sl}(2,\C)$.
Under this isomorphism: $\bm H\mapsto 2\ii\bm T_3$,
$\bm E\mapsto \ii\bm T_1-\bm T_2$,
$\bm F\mapsto \ii\bm T_1+\bm T_2$.
\end{Thm}
\end{frame}

\section{Classification of Irreducible Representations}

\begin{frame}{Representation Theory: The Basic Notions}
\begin{Def}{Definition (Lie algebra representation)}
A \textbf{representation} of a Lie algebra $\g$ on a vector space $V$
is a Lie algebra homomorphism $\pi:\g\to\gl(V)$, i.e., a linear map
preserving the bracket:
\[
\pi\Br{\SqBr{X,Y}}=\SqBr{\pi(X),\pi(Y)}.
\]
\end{Def}

\pause
\begin{Def}{Definition (Lie group representation)}
A \textbf{representation} of a Lie group $G$ on an $\F$-vector space $V$
is a smooth map $\Pi:G\to\GL(V)$ satisfying:
\begin{enumerate}
\item $\forall g,h\in G:\;\Pi(gh)=\Pi(g)\circ\Pi(h)$,
\item $\Pi(e)=I_V$.
\end{enumerate}
Equivalently, $\Pi$ is a Lie group homomorphism from $G$ to $\GL(V)$.
\end{Def}

\pause
\begin{Def}{Definition (irreducible representation)}
A representation $(\pi,V)$ is \textbf{irreducible} if the only
$\pi(\g)$-invariant subspaces of $V$ are $\set{\bm0}$ and $V$.
\end{Def}

\pause
\vspace{0.5em}
\textbf{Goal.} Classify \emph{all} finite-dimensional irreducible
representations of $\su(2)$ (equivalently, of $\mathfrak{sl}(2,\C)$).
This classification is the mathematical origin of spin.
\end{frame}

\begin{frame}{Classification of Irreps of \texorpdfstring{$\mathfrak{sl}(2,\C)$}{sl(2;C)}}
\begin{lemma}[Raising/Lowering]
Let $(\pi,V)$ be a representation of $\mathfrak{sl}(2,\C)$.
If $\bm v$ is an eigenvector of $\pi(\bm H)$ with eigenvalue $\lambda$,
then $\FUNC{\pi(\bm E)}{\bm v}$ has eigenvalue $\lambda+2$  and
$\FUNC{\pi(\bm F)}{\bm v}$ has eigenvalue $\lambda-2$ .
\end{lemma}

\pause
\begin{Thm}{Theorem (classification of finite-dimensional irreps)}
Every finite-dimensional irreducible representation of $\mathfrak{sl}(2,\C)$
is characterised by its \textbf{highest weight} $n\in\Z_{\ge0}$:
\begin{itemize}
\item $\dim V = n+1$.
\item $V$ has a basis $\set{\bm v_0,\bm v_1,\dots,\bm v_n}$ where
$\bm v_k\df\FUNC{\pi(\bm F)^k}{\bm v_0}$.
\item $\FUNC{\pi(\bm H)}{\bm v_k}=(n-2k)\bm v_k$ ; the weights are $n,n-2,\dots,-n$.
\item $\FUNC{\pi(\bm E)}{\bm v_k}=k(n-k+1)\bm v_{k-1}$.
\end{itemize}
\end{Thm}

\pause
\vspace{0.3em}
\textbf{Translation to physics.} Set $n=2j$ for $j\in\set{0,\tfrac12,1,\tfrac32,\dots}$.
Then $\dim V = 2j+1$ and the weights are $2j,2j-2,\dots,-2j$.
\end{frame}

\begin{frame}{Integration to \texorpdfstring{$\SU(2)$}{SU(2)} and the Descent Criterion}
\begin{Thm}{Theorem (integration to \texorpdfstring{$\SU(2)$}{SU(2)})}
Every finite-dimensional irreducible representation of $\su(2)$
exponentiates uniquely to an irreducible representation of $\SU(2)$:
\[
\Pi_n\Br{e^{\bm X}}\df e^{\pi_n(\bm X)}.
\]
This works because $\SU(2)\cong S^3$ is simply connected.
\end{Thm}

\pause
\begin{Thm}{Theorem (descent criterion)}
Let $\Pi_n:\SU(2)\to\GL(V)$ be the irrep of highest weight $n$.
\begin{enumerate}
\item $\Pi_n\Br{-\bm I_2}=(-1)^n\,\id_V$.
\item If $n$ is \textbf{even}, $\Pi_n$ factors through
$\Phi:\SU(2)\to\SO(3)$: gives an \textbf{ordinary representation}
of $\SO(3)$.
\item If $n$ is \textbf{odd}, $\Pi_n$ does \emph{not} descend:
gives a \textbf{projective (spinor) representation} of $\SO(3)$.
\end{enumerate}
\end{Thm}

\pause
\vspace{0.3em}
\textbf{Punchline.} $j$ integer $\leftrightarrow$ ordinary reps of $\SO(3)$.
$j$ half-integer $\leftrightarrow$ spinor reps (projective reps of $\SO(3)$).
\end{frame}

\section{Quantum Spin}

\begin{frame}{Spin Operators and Quantum Spin}
\begin{Def}{Definition (spin operators)}
Let $\Pi_n:\SU(2)\to\GL(V)$ be the irrep of highest weight $n=2j$.
The \textbf{spin operators} $S_k:\R\to\GL(V)$ are
\[
S_k\df\ii\hbar\;\dd\Pi_n(\bm T_k),\qquad k=1,2,3.
\]
\end{Def}

\pause
\begin{block}{Commutation relations}
$\SqBr{S_i,S_j}=\ii\hbar\,\eps_{ijk}S_k$: the canonical angular
momentum algebra.
\end{block}

\pause
\begin{block}{Casimir operator}
$S^2\df S_1^2+S_2^2+S_3^2$ satisfies
$S^2=\hbar^2\,j(j+1)\,\id_V$,
so states in the irrep have total spin $\hbar^2 j(j+1)$.
\end{block}

\pause
\vspace{0.3em}
\textbf{Interpretation.} The irrep of highest weight $n=2j$ describes
a quantum particle of spin $j$. The eigenvalues of $S_3$ are
$m\hbar$ for $m\in\set{-j,-j+1,\dots,j}$: the $2j+1$ spin projections.
\end{frame}

\begin{frame}{Why Half-Integer Spin?}
\begin{block}{Topological origin}
The universal cover $\SU(2)\to\SO(3)$ is \textbf{two-to-one}.
Representations of $\SO(3)$ must satisfy
$\Pi(\bm R)=\Pi(-\bm R)$ for all rotations $\bm R$.
But in $\SU(2)$, the element $-\bm I_2$ acts as $(-1)^n\id_V$.

\pause
\vspace{0.5em}
For \textbf{half-integer} $j$ ($n$ odd):
$\Pi_n\Br{-\bm I_2}=-\id_V\neq\id_V$, so these are genuine representations
of $\SU(2)$ that are only \emph{projective} representations of $\SO(3)$.
\end{block}

\pause
\begin{block}{The \texorpdfstring{$4\pi$}{4π} periodicity}
A rotation by $2\pi$ about any axis acts as $-\id_V$ on a half-integer
spin representation. A rotation by $4\pi$ is needed to return to the
identity. This is observed experimentally in neutron interferometry.
\end{block}
\end{frame}

\section{Summary}

\begin{frame}{What We Have Built}
\begin{enumerate}
\item \textbf{Topological spaces} $\to$ \textbf{smooth manifolds}: the
geometric foundation.

\pause
\item \textbf{Germs and stalks}: local behaviour of smooth maps.
\textbf{Tangent spaces} via derivations on germs.

\pause
\item \textbf{Smooth vector fields}: global derivations on $C^\infty(M)$.
\textbf{Evaluation at a point} recovers tangent vectors.

\pause
\item \textbf{Lie groups} = smooth manifolds + groups.\\
\textbf{Lie algebras} = tangent spaces at the identity.

\pause
\item \textbf{Matrix Lie groups}: $\SO(3)$ (rotations) and $\SU(2)$ (unitary $2\times2$).

\pause
\item $\SU(2)$ is a \textbf{double cover} of $\SO(3)$.
Their Lie algebras are isomorphic: $\so(3)\cong\su(2)$.

\pause
\item \textbf{Complexification}: $\su(2)_\C\cong\mathfrak{sl}(2,\C)$.
\textbf{Cartan-Weyl basis} enables the classification.

\pause
\item \textbf{Classification of irreps}: for each $j\in\set{0,\tfrac12,1,\dots}$,
a unique $(2j+1)$-dimensional irrep.

\pause
\item \textbf{Descent criterion}: $j$ integer $\to$ ordinary reps of $\SO(3)$;
$j$ half-integer $\to$ spinor (projective) reps.

\pause
\item \textbf{Quantum spin}: spin operators $S_k$, Casimir $S^2$,
$m\hbar$ eigenvalues, $4\pi$ periodicity.
\end{enumerate}
\end{frame}
